First-time setup (5 minutes)
  1. Make a free project at supabase.com.
  2. Open SQL Editor, paste the block below, press Run.
  3. Open Project Settings → API. Copy the Project URL and the anon public key into the fields above.
  4. Press Generate for a workspace key, then Turn on sync.
  5. On your other devices, open the same page, press Copy settings here and Paste settings there — or paste the copied line into any of the three fields — and turn sync on. What you type is kept if you close this sheet part-way through.
create table if not exists shipshape_state (
  id text primary key,
  data jsonb not null,
  updated_at timestamptz not null default now()
);

alter table shipshape_state enable row level security;
-- no policies: the anon key alone cannot read or list this table

create or replace function shipshape_pull(k text)
returns jsonb language sql security definer as $$
  select data from shipshape_state where id = k;
$$;

create or replace function shipshape_push(k text, d jsonb)
returns timestamptz language plpgsql security definer as $$
declare ts timestamptz;
begin
  insert into shipshape_state (id, data, updated_at)
  values (k, d, now())
  on conflict (id) do update
    set data = excluded.data, updated_at = now()
  returning updated_at into ts;
  return ts;
end $$;

revoke all on function shipshape_pull(text) from public;
revoke all on function shipshape_push(text, jsonb) from public;
grant execute on function shipshape_pull(text) to anon;
grant execute on function shipshape_push(text, jsonb) to anon;
Optional: sync item by item

Without this, each device sends its whole list on every change, and a device that has not caught up can overwrite a change made elsewhere. Run this and the app switches by itself to sending only what changed. It keeps working if you never run it.

create table if not exists shipshape_items (
  ws         text   not null,
  id         text   not null,
  kind       text   not null,
  data       jsonb,
  u          bigint not null,
  updated_at timestamptz not null default now(),
  primary key (ws, id)
);

alter table shipshape_items enable row level security;
-- no policies: the anon key alone cannot read or list this table

create or replace function shipshape_pull_items(k text, since bigint)
returns table (id text, kind text, data jsonb, u bigint)
language sql security definer as $$
  select id, kind, data, u from shipshape_items where ws = k and u > since;
$$;

create or replace function shipshape_push_items(k text, items jsonb)
returns bigint language plpgsql security definer as $$
declare it jsonb;
begin
  for it in select * from jsonb_array_elements(items) loop
    insert into shipshape_items (ws, id, kind, data, u, updated_at)
    values (k, it->>'id', it->>'kind',
            case when coalesce((it->>'deleted')::boolean, false) then null else it->'data' end,
            (it->>'u')::bigint, now())
    on conflict (ws, id) do update
      set data = excluded.data, kind = excluded.kind, u = excluded.u, updated_at = now()
      where shipshape_items.u <= excluded.u;
  end loop;
  return (extract(epoch from now()) * 1000)::bigint;
end $$;

revoke all on function shipshape_pull_items(text, bigint) from public;
revoke all on function shipshape_push_items(text, jsonb) from public;
grant execute on function shipshape_pull_items(text, bigint) to anon;
grant execute on function shipshape_push_items(text, jsonb) to anon;

create table if not exists shipshape_push_subs (
  ws         text not null,
  endpoint   text not null,
  keys       jsonb not null,
  created_at timestamptz not null default now(),
  primary key (ws, endpoint)
);

alter table shipshape_push_subs enable row level security;
-- no policies: the anon key alone cannot read or list this table

create or replace function shipshape_push_sub(k text, endpoint text, keys jsonb)
returns void language sql security definer as $$
  insert into shipshape_push_subs (ws, endpoint, keys)
  values (k, endpoint, keys)
  on conflict (ws, endpoint) do update set keys = excluded.keys;
$$;

create or replace function shipshape_push_unsub(k text, endpoint text)
returns void language sql security definer as $$
  delete from shipshape_push_subs where ws = k and endpoint = shipshape_push_unsub.endpoint;
$$;

revoke all on function shipshape_push_sub(text, text, jsonb) from public;
revoke all on function shipshape_push_unsub(text, text) from public;
grant execute on function shipshape_push_sub(text, text, jsonb) to anon;
grant execute on function shipshape_push_unsub(text, text) to anon;
